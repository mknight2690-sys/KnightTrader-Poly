  """Train transformer + policy on synthetic market data."""
  import torch
  import torch.nn as nn
  import numpy as np
  import os
  from core.systems import MarketTransformerEncoder, DeepActorCriticPolicy

  # Generate synthetic training data
  np.random.seed(42)
  n_samples = 500
  seq_len = 30
  feature_dim = 7

  # Generate sequences with different regimes
  # Bull regime: upward trend + volatility
  # Bear regime: downward trend + volatility
  # Neutral regime: flat
  sequences = []
  targets = []

  for i in range(n_samples):
      if i < n_samples // 3:
          # Bull: upward drift
          trend = 0.001
          regime = 0.5  # positive alloc
      elif i < 2 * n_samples // 3:
          # Bear: downward drift
          trend = -0.001
          regime = -0.5  # negative alloc
      else:
          # Neutral: flat
          trend = 0.0
          regime = 0.0

      # Generate sequence
      seq = np.random.randn(seq_len, feature_dim).astype(np.float32)
      # Add trend to log_ret feature
      seq[:, 6] += trend  # log_ret is the last feature
      sequences.append(seq)
      targets.append(regime)

  X = np.array(sequences)  # (n_samples, seq_len, feature_dim)
  y = np.array(targets)    # (n_samples,)

  # Create models
  transformer = MarketTransformerEncoder(feature_dim=feature_dim, d_model=64)
  policy = DeepActorCriticPolicy(state_dim=32)

  # Train
  X_tensor = torch.tensor(X, dtype=torch.float32)
  y_tensor = torch.tensor(y, dtype=torch.float32)

  optimizer = torch.optim.Adam(
      list(transformer.parameters()) + list(policy.parameters()),
      lr=0.001
  )
  criterion = nn.MSELoss()

  batch_size = 32
  n_epochs = 20

  for epoch in range(n_epochs):
      # Shuffle
      perm = torch.randperm(n_samples)
      epoch_loss = 0

      for i in range(0, n_samples, batch_size):
          batch_idx = perm[i:i+batch_size]
          batch_X = X_tensor[batch_idx]
          batch_y = y_tensor[batch_idx]

          optimizer.zero_grad()
          embedding = transformer(batch_X)
          alloc, _ = policy(embedding)
          loss = criterion(alloc.squeeze(-1), batch_y)
          loss.backward()
          optimizer.step()
          epoch_loss += loss.item() * len(batch_idx)

      if epoch % 5 == 0:
          print(f"Epoch {epoch}: loss={epoch_loss/n_samples:.4f}")

  # Save models
  os.makedirs("models", exist_ok=True)
  torch.save(transformer.state_dict(), "models/transformer.pt")
  torch.save(policy.state_dict(), "models/policy.pt")
  print("Models saved to models/transformer.pt and models/policy.pt")

  # Verify
  transformer.eval()
  policy.eval()
  with torch.no_grad():
      test_input = X_tensor[:5]
      emb = transformer(test_input)
      alloc, _ = policy(emb)
      print(f"Sample allocations: {alloc.squeeze(-1).tolist()}")
